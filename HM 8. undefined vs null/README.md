| №   | Сцена                                                                                                                                                       | Что вывелось     | Кто виноват                                                                             | Как избежать                                          |
| --- | ----------------------------------------------------------------------------------------------------------------------------------------------------------- | ---------------- | --------------------------------------------------------------------------------------- | ----------------------------------------------------- |
| 1   | let name; console.log(name);                                                                                                                                | undefined        | нет значения у переменной                                                               | объявить значение у переменной                        |
| 2   | let name = null; console.log(name);                                                                                                                         | null             | присвоили занчению null                                                                 | присвоить любое другое значение                       |
| 3   | function greet(user) { console.log('Hello,', user); } greet();                                                                                              | Hello, undefined | не дали никакого значения для функции greet()                                           | дать значение для greet()                             |
| 4   | function calculate(a, b) { return a + b; // забули повернути результат окремої змінної } console.log(calculate(2, 3));                                      | 5                | вы видется просто число 5                                                               | все отлично, можно поменять число                     |
| 5   | function calculate(a, b) { let result = a + b; // не написали return } console.log(calculate(2, 3));                                                        | undefined        | надо return для возвращение результата в функцию                                        | написать return                                       |
| 6   | function calculate(a, b) { return; } console.log(calculate(2, 3))                                                                                           | undefined        | написали return но не написали что получается запускаем функцию и возвращаем пустоту    | написать что хотим вернуть хотя бы                    |
| 7   | const user = { firstName: 'John', lastName: 'Doe' }; console.log(user.age);                                                                                 | undefined        | у нас нет значения age в объекте                                                        | присвоить значение age                                |
| 8   | const user = { firstName: 'John', middleName: null, // розробник свідомо сказав, що середнього імені немає lastName: 'Doe' }; console.log(user.middleName); | null             | у нас middleName это null                                                               | можно присвоить не null а другое значение             |
| 9   | const arr = [10, 20, 30]; console.log(arr[10]);                                                                                                             | undefined        | мы вызываем символ под индексом 10 но у нас нет символа под индексом 10, только 0, 1, 2 | если хотим вызватьсам символ 10, надо написать arr[0] |
| 10  | const user = { name: 'Anna', address: { city: 'Kyiv' } }; console.log(user.address.street);                                                                 | undefined        | у нас нет в address значения street                                                     | можно написать значение street и спокойно его вызвать |
